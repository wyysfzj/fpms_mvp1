## Components to implement (EXACT)
Add to `auth/api.py`:

```python
from app.api.deps import get_current_user
from app.modules.auth.models import T_User  
from app.modules.auth.schemas import MeResponse
from app.modules.rbac.models import T_UserRole
from app.modules.rbac.service import get_user_permissions


@router.get("/me", response_model=MeResponse)
def get_me(
    current_user: T_User = Depends(get_current_user),
    db: Session = Depends(get_db)
) -> MeResponse:
    """Get current authenticated user with roles and permissions."""
    # Get roles
    user_roles = db.query(T_UserRole).filter(T_UserRole.user_id == current_user.id).all()
    role_ids = [ur.role_id for ur in user_roles]
    
    # Get role codes
    from app.modules.auth.models import T_Role
    roles = db.query(T_Role).filter(T_Role.id.in_(role_ids)).all() if role_ids else []
    role_codes = [r.code for r in roles]
    
    # Get permissions
    perms = get_user_permissions(db, current_user.id)
    
    return MeResponse(
        user={
            "id": current_user.id,
            "username": current_user.username,
            "display_name": current_user.display_name or current_user.username
        },
        roles=role_codes,
        permissions=list(perms)
    )
```
