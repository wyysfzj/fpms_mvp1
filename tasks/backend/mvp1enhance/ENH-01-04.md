```python
from uuid import uuid4
from app.modules.auth.models import T_Role
from app.modules.rbac.models import T_RolePerm

# Permission definitions from docs/02_permissions_rbac.md
ROLE_PERMISSIONS = {
    "Admin": [
        "Auth.Login", "User.Manage", "Role.Manage",
        "Case.Read", "Case.Create", "Case.Edit", "Case.EditLimited", "Case.Export",
        "Doc.Read", "Doc.Create", "Doc.Edit", "Doc.Attach", "Doc.TemplateRender",
        "Task.Read", "Task.Create", "Task.Edit", "Task.Close", "Task.Reopen", "Task.Assign",
        "Fee.Read", "Fee.Draft.Create", "Fee.Draft.Edit", "Fee.Rate.Manage",
        "Bill.Read", "Bill.CreateFromDraft", "Bill.CreateManual", "Bill.Print",
        "Payment.Create", "Payment.Offset",
        "Client.Manage", "Template.Manage", "SystemParam.Manage"
    ],
    "Formalities": [
        "Auth.Login", "Case.Read", "Case.Create", "Case.Edit",
        "Doc.Read", "Doc.Create", "Doc.Edit", "Doc.Attach", "Doc.TemplateRender",
        "Task.Read", "Task.Create", "Task.Edit", "Task.Close", "Task.Reopen", "Task.Assign",
        "Fee.Read", "Fee.Draft.Create", "Fee.Draft.Edit", "Fee.Rate.Manage",
        "Bill.Read", "Client.Manage"
    ],
    "Agent": [
        "Auth.Login", "Case.Read", "Case.EditLimited",
        "Doc.Read", "Doc.Create", "Doc.Attach",
        "Task.Read", "Task.Create", "Task.Edit", "Task.Close", "Task.Reopen", "Task.Assign",
        "Fee.Read", "Bill.Read", "Client.Manage"
    ],
    "Finance": [
        "Auth.Login", "Case.Read",
        "Doc.Read", "Task.Read",
        "Fee.Read", "Bill.Read", "Bill.CreateFromDraft", "Bill.CreateManual", "Bill.Print",
        "Payment.Create", "Payment.Offset", "Client.Manage"
    ]
}

def seed_default_roles_perms(db: Session) -> None:
    """Seed default roles and their permissions. Idempotent."""
    for role_code, perm_codes in ROLE_PERMISSIONS.items():
        # Get or create role
        role = db.query(T_Role).filter(T_Role.code == role_code).first()
        if not role:
            role = T_Role(
                id=str(uuid4()),
                code=role_code,
                name=role_code
            )
            db.add(role)
            db.flush()
        
        # Get existing permissions
        existing = db.query(T_RolePerm).filter(T_RolePerm.role_id == role.id).all()
        existing_codes = {ep.perm_code for ep in existing}
        
        # Add missing ones
        for perm_code in perm_codes:
            if perm_code not in existing_codes:
                rp = T_RolePerm(
                    id=str(uuid4()),
                    role_id=role.id,
                    perm_code=perm_code
                )
                db.add(rp)
    
    db.commit()
```
