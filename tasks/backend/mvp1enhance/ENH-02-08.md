```python
from uuid import uuid4
from datetime import datetime
from app.modules.cases.models import Case, T_CaseApplicant, T_CaseInventor, T_Priority
from app.modules.cases.schemas import CaseCreate


def create_case(db: Session, data: CaseCreate, user_id: str) -> Case:
    """Create new case with applicants, inventors, priorities."""
    # Check case_no uniqueness
    existing = db.query(Case).filter(Case.case_no == data.case_no).first()
    if existing:
        raise_business_error(
            "CASE_NO_DUPLICATE",
            f"Case number '{data.case_no}' already exists",
            status_code=400
        )
    
    # Validate applicants
    applicants_dict = [a.model_dump() for a in data.applicants]
    validate_applicants(applicants_dict)
    
    # Create case
    case = Case(
        id=str(uuid4()),
        case_no=data.case_no,
        case_type=data.case_type,
        patent_category=data.patent_category,
        flow_dir=data.flow_dir,
        client_id=data.client_id,
        title_cn=data.title_cn,
        title_en=data.title_en,
        app_no=data.app_no,
        status="NOT_FILED",
        created_by=user_id,
        updated_by=user_id
    )
    db.add(case)
    db.flush()
    
    # Add applicants
    for app in data.applicants:
        applicant = T_CaseApplicant(
            id=str(uuid4()),
            case_id=case.id,
            seq=app.seq,
            is_first=app.is_first,
            name_cn=app.name_cn,
            name_en=app.name_en,
            address_cn=app.address_cn,
            address_en=app.address_en
        )
        db.add(applicant)
    
    # Add inventors
    for inv in data.inventors:
        inventor = T_CaseInventor(
            id=str(uuid4()),
            case_id=case.id,
            seq=inv.seq,
            name_cn=inv.name_cn,
            name_en=inv.name_en
        )
        db.add(inventor)
    
    # Add priorities
    for prio in data.priorities:
        priority = T_Priority(
            id=str(uuid4()),
            case_id=case.id,
            seq=prio.seq,
            country_code=prio.country_code,
            prio_no=prio.prio_no,
            prio_date=prio.prio_date
        )
        db.add(priority)
    
    db.commit()
    db.refresh(case)
    return case
```
