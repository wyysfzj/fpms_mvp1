    """Case inventor (one or more per case)."""
    __tablename__ = "t_case_inventor"
    
    case_id: Mapped[str] = mapped_column(String(36), ForeignKey("t_case.id", ondelete="CASCADE"), nullable=False, index=True)
    seq: Mapped[int] = mapped_column(Integer, nullable=False)
    name_cn: Mapped[str | None] = mapped_column(String(200), nullable=True)
    name_en: Mapped[str | None] = mapped_column(String(200), nullable=True)
    
    __table_args__ = (
        UniqueConstraint('case_id', 'seq', name='uq_case_inventor_seq'),
    )
