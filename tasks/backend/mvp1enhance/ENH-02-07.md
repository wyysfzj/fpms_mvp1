```python
from sqlalchemy.orm import Session
from sqlalchemy import or_
from app.core.pagination import offset_limit, PageResult
from app.modules.cases.models import Case
from app.modules.cases.schemas import CaseListItem


def list_cases(
    db: Session,
    q: str | None = None,
    client_id: str | None = None,
    status: str | None = None,
    page: int = 1,
    page_size: int = 20
) -> PageResult[CaseListItem]:
    """List cases with pagination and filters."""
    query = db.query(Case)
    
    # Search filter
    if q:
        query = query.filter(
            or_(
                Case.case_no.contains(q),
                Case.title_cn.contains(q),
                Case.title_en.contains(q),
                Case.app_no.contains(q)
            )
        )
    
    # Filters
    if client_id:
        query = query.filter(Case.client_id == client_id)
    if status:
        query = query.filter(Case.status == status)
    
    # Count total
    total = query.count()
    
    # Paginate
    off, lim = offset_limit(page, page_size)
    items = query.order_by(Case.created_at.desc()).offset(off).limit(lim).all()
    
    # Convert to schema
    list_items = [
        CaseListItem(
            id=c.id,
            case_no=c.case_no,
            case_type=c.case_type,
            patent_category=c.patent_category,
            client_id=c.client_id,
            title_cn=c.title_cn,
            title_en=c.title_en,
            status=c.status
        )
        for c in items
    ]
    
    return PageResult(items=list_items, page=page, page_size=page_size, total=total)
```
